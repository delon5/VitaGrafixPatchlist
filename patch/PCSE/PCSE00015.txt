# Shinobido 2: Revenge of Zen [US]
#
# PhyreEngine + Havok. The scene is drawn straight into the 960x544 display
# buffers (pitch 1024, three buffers of 3 MiB, one display-sized DF32 depth
# buffer of 3 MiB); there is no separate scene target, so @FB is the option that
# lowers the render resolution. Every display-derived value (context viewport,
# region clip, target size) is copied from the display table at 0x81512754,
# which the gxm init fills with immediates. Pitch, buffer sizes and the depth
# stride (1024) stay: they cover every framebuffer size below 960x544.
# Frame pacing: main() stores the target frame rate (30) in the app object;
# the run loop waits for a scheduled time (1/fps) and 60/fps vblanks on top of
# the flip's own sceDisplayWaitVblankStart. Game time is measured (8-frame
# average), capped by a "base period"; the fixed-step modes use a 1/30 base.
# Offsets are for module NID 0xF4A4F7C3 (eboot.bin, Ninja2PSP2_US).

[PCSE00015, eboot.bin, 0xF4A4F7C3]
@FB

0:0x14CB54 t2_mov(1, 1, fb_h)
# display table +0xEC width (movs.w r1, #960) / +0xEE height (movs.w r1, #544): source of viewport, clip and target size
0:0x14CBC6 t2_mov(1, 1, fb_w)
0:0x14CBCE t2_mov(1, 1, fb_h)
