# Undertale [US]
#
# GameMaker: Studio 1.4 title on the YoYo Games "PSVitaRunner". The game is
# drawn into a 640x480 application surface (size from game.win) which the
# runner then fits, aspect-correct, into a 960x544 display target. The
# runner keeps the display size in two data words (seg1+0x4194/0x4198,
# read by display_get_width/height, the surface fit rectangle, the GUI
# projection and the input mapping); @FB moves those words plus the
# display render target, colour/depth surfaces and sceDisplaySetFrameBuf.
# Display buffers keep their original allocation (pitch 1024, 0x240000).
# @MSAA rewrites the two sceGxmCreateRenderTarget parameter blocks
# (scenesPerFrame | multisampleMode << 16 stored as one word from a free
# register), sets the colour surfaces to MSAA downscale, scales the depth
# buffers (memory x2/x4, stride x2 for 4x) and passes the mode to the
# three sceGxmShaderPatcherCreateFragmentProgram sites.
# The display depth stride (0x201526) is patched only by @MSAA: 960 samples
# (>= every selectable fb_w, 32-aligned, and the untouched 0x1FE000 buffer
# already covers 960 x 544) so @FB does not need to touch it; each offset is
# patched by exactly one line because overlapping injections are refused.
# Offsets are for module NID 0x5881A7E7.

[PCSE01116, eboot.bin, 0x5881A7E7]
@FB
# sceDisplaySetFrameBuf(): movs.w r2, #960 / movs.w r3, #544 (pitch 1024 stays)
0:0x200744 t2_mov(1, 2, fb_w)
0:0x200748 t2_mov(1, 3, fb_h)
# display sceGxmCreateRenderTarget params: movs.w r1, #960 (width) / movs.w r1, #544 (height)
0:0x2013C0 t2_mov(1, 1, fb_w)
0:0x2013D0 t2_mov(1, 1, fb_h)
# display sceGxmColorSurfaceInit (x3 buffers): movs.w r1, #960 (width) / movs.w r2, #544 (height), stride 1024 stays
0:0x201496 t2_mov(1, 1, fb_w)
0:0x20149C t2_mov(1, 2, fb_h)
# display sceGxmDepthStencilSurfaceInit stride (0x201526) and memory 0x1FE000 stay: 960 samples >= every fb_w (see @MSAA)
# runner display width/height globals (display_get_width/height, surface fit rect, GUI, input mapping)
1:0x4194 uint32(fb_w)
1:0x4198 uint32(fb_h)
